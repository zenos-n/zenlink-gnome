cmake_minimum_required(VERSION 3.20)
project(quest-workspace VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(PkgConfig REQUIRED)
pkg_check_modules(HOST REQUIRED IMPORTED_TARGET gio-2.0 gio-unix-2.0 libsoup-3.0 json-glib-1.0 libpng libturbojpeg gstreamer-app-1.0 gstreamer-video-1.0)
add_executable(quest-workspace-compositor-ready host/compositor_guard.cpp)
install(TARGETS quest-workspace-compositor-ready RUNTIME DESTINATION bin)
add_executable(zenlink-workspace-configure host/configure.cpp)
target_link_libraries(zenlink-workspace-configure PRIVATE PkgConfig::HOST)
install(TARGETS zenlink-workspace-configure RUNTIME DESTINATION bin)
add_executable(zenlink-workspace-connect host/transport.cpp)
target_link_libraries(zenlink-workspace-connect PRIVATE PkgConfig::HOST)
target_compile_options(zenlink-workspace-connect PRIVATE -Wall -Wextra -Wpedantic)
install(TARGETS zenlink-workspace-connect RUNTIME DESTINATION bin)
add_executable(quest-workspace-host host/main.cpp)
target_link_libraries(quest-workspace-host PRIVATE PkgConfig::HOST)
target_compile_options(quest-workspace-host PRIVATE -Wall -Wextra -Wpedantic)
install(TARGETS quest-workspace-host RUNTIME DESTINATION bin)
pkg_check_modules(OPENVR REQUIRED IMPORTED_TARGET openvr)
add_executable(quest-workspace-pcvr host/pcvr.cpp)
target_link_libraries(quest-workspace-pcvr PRIVATE PkgConfig::HOST PkgConfig::OPENVR)
target_compile_options(quest-workspace-pcvr PRIVATE -Wall -Wextra -Wpedantic)
install(TARGETS quest-workspace-pcvr RUNTIME DESTINATION bin)
pkg_check_modules(SETTINGS REQUIRED IMPORTED_TARGET libadwaita-1)
add_executable(quest-workspace-settings host/settings.cpp)
target_link_libraries(quest-workspace-settings PRIVATE PkgConfig::HOST PkgConfig::SETTINGS)
target_compile_options(quest-workspace-settings PRIVATE -Wall -Wextra -Wpedantic)
install(TARGETS quest-workspace-settings RUNTIME DESTINATION bin)
pkg_check_modules(OVERLAY REQUIRED IMPORTED_TARGET openxr gl x11)
add_executable(quest-workspace-overlay host/overlay.cpp)
target_link_libraries(quest-workspace-overlay PRIVATE PkgConfig::HOST PkgConfig::OPENVR PkgConfig::OVERLAY)
target_compile_options(quest-workspace-overlay PRIVATE -Wall -Wextra -Wpedantic -Wno-missing-field-initializers)
install(TARGETS quest-workspace-overlay RUNTIME DESTINATION bin)

include(CTest)
if(BUILD_TESTING)
  add_executable(test-virtual-monitor tests/virtual_monitor.cpp)
  target_link_libraries(test-virtual-monitor PRIVATE PkgConfig::HOST)
  target_compile_options(test-virtual-monitor PRIVATE -UNDEBUG)
  add_executable(test-stream-settings tests/stream_settings.cpp)
  target_link_libraries(test-stream-settings PRIVATE PkgConfig::HOST)
  target_compile_options(test-stream-settings PRIVATE -UNDEBUG)
  add_test(NAME stream-settings COMMAND test-stream-settings)
  add_executable(test-compositor-guard tests/compositor_guard.cpp)
  target_compile_options(test-compositor-guard PRIVATE -UNDEBUG)
  add_test(NAME compositor-guard COMMAND test-compositor-guard)
  add_executable(test-overlay tests/overlay.cpp)
  add_test(NAME overlay-math COMMAND test-overlay)
  add_executable(test-overlay-xr tests/overlay_xr.cpp)
  target_link_libraries(test-overlay-xr PRIVATE PkgConfig::OVERLAY PkgConfig::HOST)
  add_executable(test-overlay-gl tests/overlay_gl.cpp)
  target_link_libraries(test-overlay-gl PRIVATE PkgConfig::OVERLAY)
  add_executable(test-tiles tests/tiles.cpp)
  pkg_check_modules(JPEG_TEST REQUIRED IMPORTED_TARGET libjpeg)
  target_link_libraries(test-tiles PRIVATE PkgConfig::HOST PkgConfig::JPEG_TEST)
  add_test(NAME lossless-tiles COMMAND test-tiles)
  add_executable(test-pcvr-settings tests/pcvr_settings.cpp)
  target_link_libraries(test-pcvr-settings PRIVATE PkgConfig::HOST)
  add_test(NAME pcvr-settings COMMAND test-pcvr-settings)
  foreach(target test-overlay test-overlay-gl test-overlay-xr test-pcvr-settings test-tiles)
    target_compile_options(${target} PRIVATE -UNDEBUG)
  endforeach()
  add_executable(test-hevc tests/hevc.cpp)
  target_link_libraries(test-hevc PRIVATE PkgConfig::HOST)
endif()
